// @generated by @frameless/react; do not edit.
import { useRef, useState } from 'react';

export function RenderOnce({ label, multiplier, visible, onTrace }) {
	const setupDone = useRef(null);
	if (setupDone.current === null) {
		setupDone.current = true;
		onTrace('setup', { runs: 1 });
	}
	const [count, setCount] = useState(1);
	const [prefix] = useState(() => `${label}:`);
	const derived = `${prefix}${count * multiplier}`;
	return (
		<div data-s1-root="">
			{!visible ? (
				<p data-branch="hidden">hidden</p>
			) : (
				<section data-scenario="s1">
					<output data-value="derived">{derived}</output>
					<button
						data-action="increment"
						onClick={() => {
							const nextCount = count + 1;
							setCount(nextCount);
							onTrace('change', { count: nextCount });
						}}
					>
						increment
					</button>
				</section>
			)}
		</div>
	);
}

// @generated by @frameless/solid; do not edit.
import { createSignal, untrack, Show } from 'solid-js';

export function RenderOnce(props) {
	untrack(() => props.onTrace('setup', { runs: 1 }));
	const [count, setCount] = createSignal(1);
	const prefix = untrack(() => `${props.label}:`);
	const derived = () => `${prefix}${count() * props.multiplier}`;
	return (
		<div data-s1-root="">
			<Show
				when={!props.visible}
				fallback={
					<section data-scenario="s1">
						<output data-value="derived">{derived()}</output>
						<button
							data-action="increment"
							onClick={() => {
								setCount(count() + 1);
								props.onTrace('change', { count: count() });
							}}
						>
							increment
						</button>
					</section>
				}
			>
				<p data-branch="hidden">hidden</p>
			</Show>
		</div>
	);
}

Keylines, and one shadow

A product UI usually separates panels with a ladder of grey fills. There is no elevation grey here: structure comes from the --ink keyline and the --surface-raised / --surface-sunk pair, and depth arrives once, as --elev-card on the window itself. A tool does not need six altitudes; it needs one cut edge you can trust.

Lime is the selection

On a page lime is a poster colour. In a tool it is the selected row and the selected target — a fill with an --ink label at 12.43:1, which is the same rule doing a different job. Selection never depends on colour alone: the selected chip also loses its tilt.

Empty states have no cut edge

A target with nothing captured gets a dashed keyline and a word, never a dimmed one. Dimming reads as a loading skeleton and dies in grayscale; the missing die-cut edge reads as "not printed yet", which is this brand's own vocabulary for a thing that is not there.

No stickers

The decorative nature stickers are absent entirely. They belong on a page, in the margins, at section seams — not in a workspace, where every pixel is doing a job and a charming distraction is just a distraction.

Studio is coming. Nothing to sign up for yet. There is deliberately no call to action on this page and no way to register interest: this brand's confidence comes from naming exactly what it does, and a scarcity device is the loudest possible contradiction of that.
